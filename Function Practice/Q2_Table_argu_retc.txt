#include <stdio.h>

int table(int n)
{
    int i, total = 0;

    for(i = 1; i <= 10; i++)
    {
        printf("%d x %d = %d\n", n, i, n * i);
        total = total + (n * i);
    }

    return total;
}

int main()
{
    int n, result;

    printf("Enter a number: ");
    scanf("%d", &n);

    result = table(n);

    printf("Total = %d", result);

    return 0;
}
