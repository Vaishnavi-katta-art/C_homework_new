#include <stdio.h>

int pattern(int n)
{
    int i, j, count = 0;

    for(i = 1; i <= n; i++)
    {
        for(j = 1; j <= i; j++)
        {
            printf("%d ", j);
            count++;
        }

        printf("\n");
    }

    return count;
}

int main()
{
    int n, result;

    printf("Enter number of rows: ");
    scanf("%d", &n);

    result = pattern(n);

    printf("Total numbers printed = %d", result);

    return 0;
}
